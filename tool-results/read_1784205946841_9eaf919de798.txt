    36→

    42→

    48→/* ──────────────────────── Helper ──────────────────────── */
    49→
    50→async function authFetch(action: string, payload: Record<string, string>) {
    51→  const res = await fetch('/api/auth', {
    52→    method: 'POST',
    53→    headers: { 'Content-Type': 'application/json' },
    54→    body: JSON.stringify({ action, ...payload }),
    55→  });
    56→  const data = await res.json();
    57→  if (!res.ok) throw new Error(data.error || 'Authentication failed');
    58→  return data;
    59→}

    62→

   147→      resetForm();

   151→      setLoading(false);
   152→    }

   175→      setLoading(false);
   176→    }
   177→  };
   178→
   179→  /* ────────────────── Resend OTP ────────────────── */
