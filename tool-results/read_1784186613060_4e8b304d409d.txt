   518→  const documents = useAppStore((s) => s.documents);

   520→  const addIssue = useAppStore((s) => s.addIssue);
   521→  const updateIssue = useAppStore((s) => s.updateIssue);

   524→  const setCurrentView = useAppStore((s) => s.setCurrentView);
   525→
   526→  /* ── Local state ── */
   527→  const [severityFilter, setSeverityFilter] = useState<string>('all');
   528→  const [statusFilter, setStatusFilter] = useState<string>('all');
   529→  const [analyzing, setAnalyzing] = useState(false);
   530→  const [analysisProgress, setAnalysisProgress] = useState(0);
   531→  const [selectedIssue, setSelectedIssue] = useState<DocIssue | null>(null);
   532→  const [dialogOpen, setDialogOpen] = useState(false);
   533→
   534→  /* ── Derived counts ── */
   535→  const counts = useMemo(() => {
   536→    const openIssues = issues.filter((i) => i.status === 'open');
   537→    return {
   538→      critical: openIssues.filter((i) => i.severity === 'critical').length,
   539→      high: openIssues.filter((i) => i.severity === 'high').length,
   540→      medium: openIssues.filter((i) => i.severity === 'medium').length,
   541→      low: openIssues.filter((i) => i.severity === 'low').length,
   542→      total: openIssues.length,
   543→    };
   544→  }, [issues]);
   545→
   546→  /* ── Filtered issues ── */
   547→  const filteredIssues = useMemo(() => {
   548→    return issues
   549→      .filter((issue) => {
   550→        if (severityFilter !== 'all' && issue.severity !== severityFilter) return false;
   551→        if (statusFilter !== 'all' && issue.status !== statusFilter) return false;
   552→        return true;
   553→      })
   554→      .sort((a, b) => {
   555→        const severityOrder: Record<SeverityLevel, number> = {
   556→          critical: 0,
   557→          high: 1,
   558→          medium: 2,
   559→          low: 3,
   560→        };
   561→        if (severityOrder[a.severity] !== severityOrder[b.severity]) {
   562→          return severityOrder[a.severity] - severityOrder[b.severity];
   563→        }
   564→        return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
   565→      });
   566→  }, [issues, severityFilter, statusFilter]);
   567→
   568→  /* ── Run AI Analysis ── */
   569→  const runAnalysis = useCallback(async () => {
   570→    if (documents.length === 0 || analyzing) return;
   571→
   572→    setAnalyzing(true);
   573→    setAnalysisProgress(10);
   574→
   575→    try {
   576→      // Simulate progress while waiting
   577→      const progressInterval = setInterval(() => {
   578→        setAnalysisProgress((prev) => {
   579→          if (prev >= 85) {
   580→            clearInterval(progressInterval);
   581→            return 85;
   582→          }
   583→          return prev + Math.random() * 12;
   584→        });
   585→      }, 600);
   586→
   587→      const response = await fetch('/api/analyze', {
   588→        method: 'POST',
   589→        headers: { 'Content-Type': 'application/json' },
   590→        body: JSON.stringify({ documents }),
   591→      });
   592→
   593→      clearInterval(progressInterval);
   594→
   595→      if (!response.ok) {
   596→        throw new Error('Analysis request failed');
   597→      }
   598→
   599→      const data = await response.json();
   600→      setAnalysisProgress(95);
   601→
   602→      const newIssues: DocIssue[] = data.issues || [];
   603→
   604→      // Add each issue to store
   605→      newIssues.forEach((issue: DocIssue) => {
   606→        addIssue(issue);
   607→      });
   608→
   609→      setAnalysisProgress(100);
   610→
   611→      // Record activity
   612→      addActivity(
   613→        'AI Analysis',
   614→        `AI Analysis completed - ${newIssues.length} issue${newIssues.length !== 1 ? 's' : ''} found`
   615→      );
   616→
   617→      // Add notification
   618→      if (newIssues.length > 0) {
   619→        const criticalCount = newIssues.filter(
   620→          (i: DocIssue) => i.severity === 'critical' || i.severity === 'high'
   621→        ).length;
   622→
   623→        addNotification({
   624→          id: crypto.randomUUID(),
   625→          type: 'system',
   626→          title: 'AI Analysis Complete',
   627→          message: `Found ${newIssues.length} issue${newIssues.length !== 1 ? 's' : ''}${
   628→            criticalCount > 0 ? ` (${criticalCount} critical/high)` : ''
   629→          } across your documents.`,
   630→          read: false,
   631→          actionUrl: 'issues',
   632→          createdAt: new Date().toISOString(),
   633→        });
   634→      } else {
   635→        addNotification({
   636→          id: crypto.randomUUID(),
   637→          type: 'system',
   638→          title: 'AI Analysis Complete',
   639→          message: 'No issues detected. Your documents look good!',
   640→          read: false,
   641→          createdAt: new Date().toISOString(),
   642→        });
   643→      }
   644→    } catch (error) {
   645→      console.error('Analysis error:', error);
   646→      addNotification({
   647→        id: crypto.randomUUID(),
   648→        type: 'system',
   649→        title: 'Analysis Failed',
   650→        message: 'Could not complete AI analysis. Please try again.',
   651→        read: false,
   652→        createdAt: new Date().toISOString(),
   653→      });
   654→    } finally {
   655→      setTimeout(() => {
   656→        setAnalyzing(false);
   657→        setAnalysisProgress(0);
   658→      }, 500);
   659→    }
   660→  }, [documents, analyzing, addIssue, addActivity, addNotification]);
   661→
   662→  /* ── Status change handler ── */
   663→  const handleStatusChange = useCallback(
   664→    (id: string, status: IssueStatus) => {
   665→      updateIssue(id, { status });
   666→    },
   667→    [updateIssue]
   668→  );
   669→
   670→  /* ── Fix handler ── */
   671→  const handleFix = useCallback((issue: DocIssue) => {
   672→    setSelectedIssue(issue);
   673→    setDialogOpen(true);
   674→  }, []);
   675→
   676→  /* ── Go to correction ── */
   677→  const goToCorrection = useCallback(() => {
   678→    setCurrentView('correction');
   679→  }, [setCurrentView]);
   680→
   681→  /* ──────────────────────── Empty State: No Documents ──────────────────────── */
   682→
   683→  if (documents.length === 0) {
   684→    return (
   685→      <div className="flex min-h-[60vh] items-center justify-center p-6">
   686→        <motion.div
   687→          initial={{ opacity: 0, scale: 0.95 }}
   688→          animate={{ opacity: 1, scale: 1 }}
   689→          className="flex max-w-md flex-col items-center text-center"
   690→        >
   691→          <div className="mb-4 flex size-20 items-center justify-center rounded-full bg-emerald-100 dark:bg-emerald-900/30">
   692→            <Shield className="size-10 text-emerald-500" />
   693→          </div>
   694→          <h2 className="mb-2 text-xl font-semibold text-foreground">
   695→            No Documents Yet
   696→          </h2>
   697→          <p className="mb-6 text-sm text-muted-foreground leading-relaxed">
   698→            Upload documents first to run AI analysis. DocSync will detect
   699→            inconsistencies, mismatches, and errors across your documents.
   700→          </p>
   701→          <Button
   702→            onClick={() => setCurrentView('documents')}
   703→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   704→          >
   705→            Upload Documents
   706→          </Button>
   707→        </motion.div>
   708→      </div>
   709→    );
   710→  }
   711→
   712→  /* ════════════════════════════════════════════════════════════════════════
   713→     Main render
   714→     ════════════════════════════════════════════════════════════════════════ */
   715→
   716→  return (
   717→    <div className="mx-auto w-full max-w-5xl space-y-6 p-4 sm:p-6">
   718→      {/* ──────── Header ──────── */}
   719→      <motion.div {...fadeInUp} transition={{ duration: 0.4 }}>
   720→        <div className="flex flex-col gap-4 sm:flex-row sm:items-start sm:justify-between">
   721→          <div>
   722→            <h1 className="flex items-center gap-2 text-2xl font-bold text-foreground">
   723→              <Shield className="size-7 text-emerald-500" />
   724→              AI Issue Detection
   725→            </h1>
   726→            <p className="mt-1 text-sm text-muted-foreground">
   727→              Scan your documents for inconsistencies, mismatches, and errors using AI-powered analysis.
   728→            </p>
   729→          </div>
   730→
   731→          <Button
   732→            onClick={runAnalysis}
   733→            disabled={analyzing}
   734→            className="shrink-0 bg-emerald-600 hover:bg-emerald-700 text-white shadow-md shadow-emerald-500/20"
   735→          >
   736→            {analyzing ? (
   737→              <>
   738→                <Loader2 className="mr-2 size-4 animate-spin" />
   739→                Analyzing...
   740→              </>
   741→            ) : (
   742→              <>
   743→                <RefreshCw className="mr-2 size-4" />
   744→                Run AI Analysis
   745→              </>
   746→            )}
   747→          </Button>
   748→        </div>
   749→
   750→        {/* Analysis progress bar */}
   751→        <AnimatePresence>
   752→          {analyzing && (
   753→            <motion.div
   754→              initial={{ opacity: 0, height: 0 }}
   755→              animate={{ opacity: 1, height: 'auto' }}
   756→              exit={{ opacity: 0, height: 0 }}
   757→              className="mt-4 overflow-hidden"
   758→            >
   759→              <div className="rounded-lg border border-emerald-200 bg-emerald-50/50 p-4 dark:border-emerald-800 dark:bg-emerald-950/20">
   760→                <div className="mb-2 flex items-center justify-between text-sm">
   761→                  <span className="font-medium text-emerald-700 dark:text-emerald-400">
   762→                    Analyzing your documents...
   763→                  </span>
   764→                  <span className="text-xs text-muted-foreground">
   765→                    {Math.round(analysisProgress)}%
   766→                  </span>
   767→                </div>
   768→                <Progress value={analysisProgress} className="h-2" />
   769→              </div>
   770→            </motion.div>
   771→          )}
   772→        </AnimatePresence>
   773→      </motion.div>
   774→
   775→      {/* ──────── Summary Cards ──────── */}
   776→      <motion.div
   777→        variants={staggerContainer}
   778→        initial="initial"
   779→        animate="animate"
   780→        className="grid grid-cols-2 gap-3 sm:gap-4 lg:grid-cols-4"
   781→      >
   782→        {/* Critical */}
   783→        <motion.div variants={staggerItem}>
   784→          <Card className="border-red-200 bg-red-50/50 dark:border-red-800 dark:bg-red-950/20">
   785→            <CardContent className="flex items-center gap-3 p-4">
   786→              <div className="flex size-10 shrink-0 items-center justify-center rounded-lg bg-red-100 dark:bg-red-900/40">
   787→                <XCircle className="size-5 text-red-600 dark:text-red-400" />
   788→              </div>
   789→              <div>
   790→                <p className="text-2xl font-bold text-red-600 dark:text-red-400">{counts.critical}</p>
   791→                <p className="text-xs font-medium text-red-500/80 dark:text-red-400/70">Critical</p>
   792→              </div>
   793→            </CardContent>
   794→          </Card>
   795→        </motion.div>
   796→
   797→        {/* High */}
   798→        <motion.div variants={staggerItem}>
   799→          <Card className="border-orange-200 bg-orange-50/50 dark:border-orange-800 dark:bg-orange-950/20">
   800→            <CardContent className="flex items-center gap-3 p-4">
   801→              <div className="flex size-10 shrink-0 items-center justify-center rounded-lg bg-orange-100 dark:bg-orange-900/40">
   802→                <AlertTriangle className="size-5 text-orange-600 dark:text-orange-400" />
   803→              </div>
   804→              <div>
   805→                <p className="text-2xl font-bold text-orange-600 dark:text-orange-400">{counts.high}</p>
   806→                <p className="text-xs font-medium text-orange-500/80 dark:text-orange-400/70">High</p>
   807→              </div>
   808→            </CardContent>
   809→          </Card>
   810→        </motion.div>
   811→
   812→        {/* Medium */}
   813→        <motion.div variants={staggerItem}>
   814→          <Card className="border-amber-200 bg-amber-50/50 dark:border-amber-800 dark:bg-amber-950/20">
   815→            <CardContent className="flex items-center gap-3 p-4">
   816→              <div className="flex size-10 shrink-0 items-center justify-center rounded-lg bg-amber-100 dark:bg-amber-900/40">
   817→                <AlertCircle className="size-5 text-amber-600 dark:text-amber-400" />
   818→              </div>
   819→              <div>
   820→                <p className="text-2xl font-bold text-amber-600 dark:text-amber-400">{counts.medium}</p>
   821→                <p className="text-xs font-medium text-amber-500/80 dark:text-amber-400/70">Medium</p>
   822→              </div>
   823→            </CardContent>
   824→          </Card>
   825→        </motion.div>
   826→
   827→        {/* Low */}
   828→        <motion.div variants={staggerItem}>
   829→          <Card className="border-teal-200 bg-teal-50/50 dark:border-teal-800 dark:bg-teal-950/20">
   830→            <CardContent className="flex items-center gap-3 p-4">
   831→              <div className="flex size-10 shrink-0 items-center justify-center rounded-lg bg-teal-100 dark:bg-teal-900/40">
   832→                <Info className="size-5 text-teal-600 dark:text-teal-400" />
   833→              </div>
   834→              <div>
   835→                <p className="text-2xl font-bold text-teal-600 dark:text-teal-400">{counts.low}</p>
   836→                <p className="text-xs font-medium text-teal-500/80 dark:text-teal-400/70">Low</p>
   837→              </div>
   838→            </CardContent>
   839→          </Card>
   840→        </motion.div>
   841→      </motion.div>
   842→
   843→      {/* ──────── Filters ──────── */}
   844→      <motion.div {...fadeInUp} transition={{ duration: 0.4, delay: 0.15 }}>
   845→        <div className="flex flex-wrap items-center gap-3">
   846→          <div className="flex items-center gap-2 text-sm font-medium text-muted-foreground">
   847→            <Filter className="size-4" />
   848→            Filters
   849→          </div>
   850→
   851→          <Select value={severityFilter} onValueChange={setSeverityFilter}>
   852→            <SelectTrigger className="w-[160px] h-9 text-sm">
   853→              <SelectValue placeholder="Severity" />
   854→            </SelectTrigger>
   855→            <SelectContent>
   856→              <SelectItem value="all">All Severities</SelectItem>
   857→              <SelectItem value="critical">Critical</SelectItem>
   858→              <SelectItem value="high">High</SelectItem>
   859→              <SelectItem value="medium">Medium</SelectItem>
   860→              <SelectItem value="low">Low</SelectItem>
   861→            </SelectContent>
   862→          </Select>
   863→
   864→          <Select value={statusFilter} onValueChange={setStatusFilter}>
   865→            <SelectTrigger className="w-[160px] h-9 text-sm">
   866→              <SelectValue placeholder="Status" />
   867→            </SelectTrigger>
   868→            <SelectContent>
   869→              <SelectItem value="all">All Statuses</SelectItem>
   870→              <SelectItem value="open">Open</SelectItem>
   871→              <SelectItem value="resolved">Resolved</SelectItem>
   872→              <SelectItem value="dismissed">Dismissed</SelectItem>
   873→            </SelectContent>
   874→          </Select>
   875→
   876→          {(severityFilter !== 'all' || statusFilter !== 'all') && (
   877→            <Button
   878→              variant="ghost"
   879→              size="sm"
   880→              className="h-9 text-xs text-muted-foreground"
   881→              onClick={() => {
   882→                setSeverityFilter('all');
   883→                setStatusFilter('all');
   884→              }}
   885→            >
   886→              Clear Filters
   887→            </Button>
   888→          )}
   889→
   890→          <div className="ml-auto text-xs text-muted-foreground">
   891→            {filteredIssues.length} issue{filteredIssues.length !== 1 ? 's' : ''}
   892→          </div>
   893→        </div>
   894→      </motion.div>
   895→
   896→      {/* ──────── Issues List ──────── */}
   897→      {issues.length === 0 ? (
   898→        /* Empty state — no issues */
   899→        <motion.div
   900→          initial={{ opacity: 0, y: 20 }}
   901→          animate={{ opacity: 1, y: 0 }}
   902→          transition={{ duration: 0.4, delay: 0.2 }}
   903→        >
   904→          <Card className="border-dashed border-emerald-300 bg-emerald-50/30 dark:border-emerald-700 dark:bg-emerald-950/10">
   905→            <CardContent className="flex flex-col items-center py-12 text-center">
   906→              <div className="mb-4 flex size-16 items-center justify-center rounded-full bg-emerald-100 dark:bg-emerald-900/30">
   907→                <CheckCircle2 className="size-8 text-emerald-500" />
   908→              </div>
   909→              <h3 className="mb-1 text-lg font-semibold text-foreground">
   910→                No issues detected!
   911→              </h3>
   912→              <p className="max-w-sm text-sm text-muted-foreground">
   913→                Your documents look good. Run AI Analysis to check for any inconsistencies,
   914→                mismatches, or errors across your documents.
   915→              </p>
   916→            </CardContent>
   917→          </Card>
   918→        </motion.div>
   919→      ) : filteredIssues.length === 0 ? (
   920→        /* No matching issues after filter */
   921→        <motion.div
   922→          initial={{ opacity: 0, y: 20 }}
   923→          animate={{ opacity: 1, y: 0 }}
   924→          transition={{ duration: 0.4, delay: 0.2 }}
   925→        >
   926→          <Card className="border-dashed">
   927→            <CardContent className="flex flex-col items-center py-12 text-center">
   928→              <Filter className="mb-3 size-8 text-muted-foreground/50" />
   929→              <h3 className="mb-1 text-base font-semibold text-foreground">
   930→                No matching issues
   931→              </h3>
   932→              <p className="text-sm text-muted-foreground">
   933→                No issues match the current filter criteria. Try adjusting your filters.
   934→              </p>
   935→            </CardContent>
   936→          </Card>
   937→        </motion.div>
   938→      ) : (
   939→        <ScrollArea className="max-h-[calc(100vh-420px)]">
   940→          <motion.div
   941→            variants={staggerContainer}
   942→            initial="initial"
   943→            animate="animate"
   944→            className="space-y-4 pb-4"
   945→          >
   946→            {filteredIssues.map((issue) => (
   947→              <IssueCard
   948→                key={issue.id}
   949→                issue={issue}
   950→                onFix={handleFix}
   951→                onStatusChange={handleStatusChange}
   952→              />
   953→            ))}
   954→          </motion.div>
   955→        </ScrollArea>
   956→      )}
   957→
   958→      {/* ──────── Fix Guidance Dialog ──────── */}
   959→      <FixGuidanceDialog
   960→        issue={selectedIssue}
   961→        open={dialogOpen}
   962→        onOpenChange={setDialogOpen}
   963→        onGoToCorrection={goToCorrection}
   964→      />
   965→    </div>
   966→  );
   967→}
