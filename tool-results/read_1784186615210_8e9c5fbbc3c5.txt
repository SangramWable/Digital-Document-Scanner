    77→/* ──────────────────────── Animated Counter ──────────────────────── */
    78→
    79→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
    80→  const [count, setCount] = useState(0);
    81→
    82→  useEffect(() => {
    83→    let start = 0;
    84→    const step = target / (duration / 16);
    85→    const timer = setInterval(() => {
    86→      start += step;
    87→      if (start >= target) {
    88→        setCount(target);
    89→        clearInterval(timer);
    90→      } else {
    91→        setCount(Math.floor(start));
    92→      }
    93→    }, 16);
    94→    return () => clearInterval(timer);
    95→  }, [target, duration]);
    96→
    97→  return <span>{count}</span>;
    98→}

   100→/* ──────────────────────── Health Score Ring ──────────────────────── */
   101→
   102→function HealthScoreRing({ score }: { score: number }) {

   104→  const circumference = 2 * Math.PI * radius;

   106→
   107→  const color = score >= 80 ? '#10b981' : score >= 50 ? '#f59e0b' : '#ef4444';
   108→  const label =
   109→    score >= 80 ? 'Excellent' : score >= 50 ? 'Needs Attention' : score > 0 ? 'Critical' : 'No Data';
   110→
   111→  return (
   112→    <div className="relative flex flex-col items-center gap-2">
   113→      <svg width="180" height="180" viewBox="0 0 180 180" className="-rotate-90">
   114→        {/* Background ring */}
   115→        <circle
   116→          cx="90"
   117→          cy="90"
   118→          r={radius}
   119→          fill="none"
   120→          stroke="currentColor"
   121→          strokeWidth="12"
   122→          className="text-muted/30"
   123→        />
   124→        {/* Progress ring */}
   125→        <motion.circle
   126→          cx="90"
   127→          cy="90"
   128→          r={radius}
   129→          fill="none"
   130→          stroke={color}

   134→          initial={{ strokeDashoffset: circumference }}

   398→/* ──────────────────────── Recent Activity Timeline ──────────────────────── */
   399→
   400→function RecentActivityTimeline() {
   401→  const activities = useAppStore((s) => s.activities);
   402→
   403→  const recentActivities = useMemo(() => activities.slice(0, 5), [activities]);
   404→
   405→  const activityIcon = (action: string) => {
   406→    const lower = action.toLowerCase();
   407→    if (lower.includes('upload')) return Upload;
   408→    if (lower.includes('issue') || lower.includes('resolved')) return FileCheck;
   409→    if (lower.includes('delete') || lower.includes('removed')) return Trash2;
   410→    if (lower.includes('login')) return Shield;
   411→    return Activity;
   412→  };
   413→
   414→  const activityColor = (action: string) => {
   415→    const lower = action.toLowerCase();
   416→    if (lower.includes('upload')) return '#10b981';
   417→    if (lower.includes('issue') || lower.includes('resolved')) return '#0d9488';
   418→    if (lower.includes('delete') || lower.includes('removed')) return '#ef4444';
   419→    if (lower.includes('login')) return '#059669';
   420→    return '#64748b';
   421→  };
   422→
   423→  const formatTimestamp = (ts: string) => {
   424→    const date = new Date(ts);
   425→    const now = new Date();
   426→    const diffMs = now.getTime() - date.getTime();
   427→    const diffMins = Math.floor(diffMs / 60000);
   428→    if (diffMins < 1) return 'Just now';
   429→    if (diffMins < 60) return `${diffMins}m ago`;
   430→    const diffHrs = Math.floor(diffMins / 60);
   431→    if (diffHrs < 24) return `${diffHrs}h ago`;
   432→    const diffDays = Math.floor(diffHrs / 24);
   433→    if (diffDays < 7) return `${diffDays}d ago`;
   434→    return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
   435→  };
   436→
   437→  return (
   438→    <motion.div variants={staggerItem}>
   439→      <Card className="border-border/40 bg-card/60 backdrop-blur-md transition-shadow duration-300 hover:shadow-lg hover:shadow-emerald-500/5">
   440→        <CardHeader>
   441→          <div className="flex items-center gap-2">
   442→            <div className="flex size-8 items-center justify-center rounded-lg bg-emerald-500/15">
   443→              <Clock className="size-4 text-emerald-600 dark:text-emerald-400" />
   444→            </div>
   445→            <CardTitle className="text-base">Recent Activity</CardTitle>
   446→          </div>
   447→          <CardDescription>Your latest actions and events</CardDescription>
   448→        </CardHeader>
   449→        <CardContent>
   450→          {recentActivities.length === 0 ? (
   451→            <div className="flex flex-col items-center justify-center py-8 text-center">
   452→              <div className="mb-3 flex size-14 items-center justify-center rounded-full bg-muted/50">
   453→                <Activity className="size-6 text-muted-foreground/40" />
   454→              </div>
   455→              <p className="text-sm text-muted-foreground">No recent activity</p>
   456→              <p className="text-xs text-muted-foreground/70">
   457→                Your actions will appear here as you use the app.
   458→              </p>
   459→            </div>
   460→          ) : (
   461→            <div className="relative space-y-0">
   462→              {recentActivities.map((act, idx) => {
   463→                const Icon = activityIcon(act.action);
   464→                const color = activityColor(act.action);
   465→                const isLast = idx === recentActivities.length - 1;
   466→                return (
   467→                  <motion.div
   468→                    key={act.id}
   469→                    initial={{ opacity: 0, x: -10 }}
   470→                    animate={{ opacity: 1, x: 0 }}
   471→                    transition={{ delay: idx * 0.08, duration: 0.3 }}

   479→                    <div
   480→                      className="relative z-10 mt-0.5 flex size-[30px] shrink-0 items-center justify-center rounded-full border-2 border-background"
   481→                      style={{ backgroundColor: color + '20' }}

   487→                      <span className="text-sm font-medium text-foreground">{act.action}</span>
   488→                      <span className="text-xs text-muted-foreground">{act.description}</span>
   489→                      <span className="text-[10px] text-muted-foreground/60">
   490→                        {formatTimestamp(act.timestamp)}
   491→                      </span>
   492→                    </div>
   493→                  </motion.div>
   494→                );
   495→              })}
   496→            </div>
   497→          )}
   498→        </CardContent>
   499→      </Card>
