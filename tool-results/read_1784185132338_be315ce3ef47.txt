    56→/* ──────────────────────────── Animated counter ──────────────────────────── */
    57→
    58→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
    59→  const ref = useRef<HTMLSpanElement>(null);
    60→  const inView = useInView(ref, { once: true, margin: '-50px' });
    61→  const [count, setCount] = useState(0);
    62→
    63→  useEffect(() => {
    64→    if (!inView) return;
    65→    let start = 0;
    66→    const duration = 1800;
    67→    const stepTime = 16;
    68→    const steps = Math.ceil(duration / stepTime);
    69→    const increment = target / steps;
    70→
    71→    const timer = setInterval(() => {
    72→      start += increment;
    73→      if (start >= target) {
    74→        setCount(target);
    75→        clearInterval(timer);
    76→      } else {
    77→        setCount(Math.floor(start));
    78→      }
    79→    }, stepTime);
    80→
    81→    return () => clearInterval(timer);
    82→  }, [inView, target]);
    83→
    84→  return (
    85→    <span ref={ref}>
    86→      {count.toLocaleString('en-IN')}
    87→      {suffix}
    88→    </span>
    89→  );
    90→}
