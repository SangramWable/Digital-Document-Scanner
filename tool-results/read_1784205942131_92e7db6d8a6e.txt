    48→    48→/* ──────────────────────── Helper ──────────────────────── */
    49→    49→
    50→    50→async function authFetch(action: string, payload: Record<string, string>) {
    51→    51→  const res = await fetch('/api/auth', {
    52→    52→    method: 'POST',
    53→    53→    headers: { 'Content-Type': 'application/json' },
    54→    54→    body: JSON.stringify({ action, ...payload }),
    55→    55→  });
    56→    56→  const data = await res.json();
    57→    57→  if (!res.ok) throw new Error(data.error || 'Authentication failed');
    58→    58→  return data;
    59→    59→}
